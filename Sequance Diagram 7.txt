sequenceDiagram
    autonumber
    actor User as User
    participant MF as :MainFrame
    participant TC as :TransactionController
    participant DB as :DatabaseManager

    User->>MF: navigates to "Reports" screen
    MF->>MF: switchPanel("Reports")
    MF-->>User: displays date range selector (defaults to current month)
    User->>MF: selects a date range (d)
    MF->>TC: FilterByDate(d)
    TC->>DB: method(Transactions)
    DB-->>TC: type

    alt No Transactions (Exceptional Scenario)
        TC-->>MF: List (empty)
        MF-->>User: displays "No transaction data available for this period."
    else Transactions Exist (Normal Scenario)
        TC-->>MF: List (populated)
        MF->>TC: FilterByCategory(c)
        TC-->>MF: List
        MF->>TC: GetIncome()
        TC-->>MF: int
        MF->>TC: GetExpense()
        TC-->>MF: int
        MF-->>User: displays pie chart (expenses by category)
        MF-->>User: displays bar chart (income vs expenses)
        MF-->>User: displays key insight message (e.g., "Food spending is 15% above average.")
    end
