sequenceDiagram
    autonumber
    actor User as User
    participant MF as :MainFrame
    participant TV as :TransactionView
    participant TC as :TransactionController
    participant DB as :DatabaseManager
    participant DC as :DashboardController
    participant DV as :DashboardView

    User->>MF: clicks "+" (Add)
    MF->>MF: switchPanel("TransactionView")
    MF-->>User: displays "Add Transaction" screen
    User->>TV: enters amount, category, type, description
    User->>TV: clicks "Save Transaction"
    TV->>TV: getAmount() : double
    TV->>TV: getCategory() : String
    TV->>TC: Add(Transactions)

    alt is Invalid Amount (Exceptional Scenario)
        TC->>TC: HandleError("Please enter an amount.")
        TC-->>TV: void
        TV-->>User: displays error message "Please enter an amount."
    else is Valid Amount (Normal Scenario)
        TC->>DB: method(Transactions)
        DB-->>TC: type
        TC->>TV: clearFields()
        TC-->>MF: void
        MF->>MF: switchPanel("DashboardView")
        MF->>DC: updateView()
        DC->>DV: updateTransactionTable(List)
        DC-->>MF: void
        MF-->>User: redirects to main dashboard with new transaction
    end
