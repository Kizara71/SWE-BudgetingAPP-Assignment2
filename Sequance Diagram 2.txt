sequenceDiagram
    actor User as Registered User
    participant UI as :LoginUI
    participant AC as :AuthController
    participant DB as :UserDB

    User->>UI: Clicks "Log In"
    UI-->>User: displays log-in page
    User->>UI: fills credentials & clicks "Log In"
    UI->>AC: Login(email, password)
    AC->>AC: Validation

    alt [is invalid (Exceptional Scenario)]
        AC->>DB: FindUserByEmail(email)
        DB-->>AC: verificationFailed()
        AC-->>UI: bool(false)
        UI-->>User: DisplayError "Invalid email or password"
    else [is valid (Normal Scenario)]
        AC->>DB: FindUserByEmail(email)
        DB-->>AC: UserRecorded(email, password)
        AC-->>UI: bool(true)
        UI->>UI: SwitchPane("DashboardView")
        UI-->>User: redirects to main dashboard
    end
